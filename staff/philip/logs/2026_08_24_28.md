```sh
docker run --rm -it --ipc=host --ulimit memlock=-1 --ulimit stack=67108864 --gpus=all \

```

### TensorRT LLM OAI Serve
```sh
trtllm-serve nvidia/Gemma-4-31B-IT-NVFP4  \
  --host 0.0.0.0 \
  --port 30000 \
  --max_seq_len 8192 \
  --max_num_tokens 8192 \
  --kv_cache_free_gpu_memory_fraction 0.8
```

### SGLang
```sh
docker run --gpus all \
    --shm-size 64g \
    -p 30000:30000 \
    -v ~/.cache/huggingface:/root/.cache/huggingface \
    --ipc=host \
    lmsysorg/sglang:latest-runtime \
    python3 -m sglang.launch_server --model-path nvidia/Gemma-4-31B-IT-NVFP4 --host 0.0.0.0 --port 30000
```

### VLLM
```sh
docker run --gpus all \
      -v ~/.cache/huggingface:/root/.cache/huggingface \
      -p 30000:30000 \
      --ipc=host \
      vllm/vllm-openai:v0.28.0 \
      --model nvidia/Gemma-4-31B-IT-NVFP4
```
