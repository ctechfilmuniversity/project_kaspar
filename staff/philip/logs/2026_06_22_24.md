### Eval Multi-Device Link: ZMQ

`sender.py`
```py
import cv2
import zmq
import pyaudio
import threading
import sys
import time

context = zmq.Context()
FORMAT = pyaudio.paInt16
CHUNK = 512


def stream_video():
    vid_socket = context.socket(zmq.PUB)
    vid_socket.setsockopt(zmq.SNDHWM, 2)
    vid_socket.bind("tcp://*:5555")
    cap = cv2.VideoCapture(0)
    while True:
        ret, frame = cap.read()
        if not ret:
            continue
        _, buffer = cv2.imencode(".jpg", frame, [cv2.IMWRITE_JPEG_QUALITY, 80])

        # Embed the exact sender creation timestamp in-band
        vid_socket.send_string(str(time.time()), flags=zmq.SNDMORE)
        vid_socket.send(buffer.tobytes())


def stream_audio():
    aud_socket = context.socket(zmq.PUB)
    aud_socket.setsockopt(zmq.SNDHWM, 2)
    aud_socket.bind("tcp://*:5556")
    p = pyaudio.PyAudio()

    try:
        device_info = p.get_default_input_device_info()
        rate = int(device_info["defaultSampleRate"])
        channels = int(device_info["maxInputChannels"])
        channels = 1 if channels >= 1 else channels
    except IOError:
        return

    stream = p.open(
        format=FORMAT, channels=channels, rate=rate, input=True, frames_per_buffer=CHUNK
    )

    while True:
        try:
            data = stream.read(CHUNK, exception_on_overflow=False)
            # Embed the exact sender creation timestamp in-band
            aud_socket.send_string(str(time.time()), flags=zmq.SNDMORE)
            aud_socket.send(data)
        except OSError:
            pass


def calibration_responder():
    # Sync socket handles the handshake protocol
    sync_socket = context.socket(zmq.REP)
    sync_socket.bind("tcp://*:5557")
    while True:
        sync_socket.recv()  # Await ping from receiver
        # Reply with the sender's absolute system time
        sync_socket.send_string(str(time.time()))


threading.Thread(target=stream_video, daemon=True).start()
threading.Thread(target=stream_audio, daemon=True).start()
threading.Thread(target=calibration_responder, daemon=True).start()

print("Calibrated streaming active. Awaiting receiver connection...")
try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    sys.exit(0)
```

`receiver.py`
```py
import cv2
import zmq
import pyaudio
import numpy as np
import threading
import sys
import time

SENDER_IP = "100.69.214.21"
context = zmq.Context()

FORMAT = pyaudio.paInt16
CHANNELS = 1
RATE = 48000
CHUNK = 512

# Global clock drift offset variable calculated at startup
clock_drift = 0.0


def calibrate_clock():
    global clock_drift
    sync_socket = context.socket(zmq.REQ)
    sync_socket.connect(f"tcp://{SENDER_IP}:5557")

    print("Calibrating clock drift...")
    rtts = []
    drifts = []

    # Run 5 iterations to eliminate initial network jitter anomalies
    for _ in range(5):
        t1 = time.time()
        sync_socket.send(b"ping")
        t_sender = float(sync_socket.recv_string())
        t2 = time.time()

        rtt = t2 - t1
        # Estimated sender time when t2 was recorded = t_sender + (rtt / 2)
        # Drift = Sender Time - Receiver Time
        drift = (t_sender + (rtt / 2.0)) - t2

        rtts.append(rtt)
        drifts.append(drift)
        time.sleep(0.1)

    clock_drift = sum(drifts) / len(drifts)
    avg_rtt_ms = (sum(rtts) / len(rtts)) * 1000
    print(
        f"Calibration Complete. Est. Network RTT: {avg_rtt_ms:.1f}ms. Clock Drift: {clock_drift:.4f}s"
    )


def receive_audio():
    aud_socket = context.socket(zmq.SUB)
    aud_socket.setsockopt(zmq.RCVHWM, 2)
    aud_socket.connect(f"tcp://{SENDER_IP}:5556")
    aud_socket.setsockopt(zmq.SUBSCRIBE, b"")

    p = pyaudio.PyAudio()
    stream = p.open(
        format=FORMAT,
        channels=CHANNELS,
        rate=RATE,
        output=True,
        frames_per_buffer=CHUNK,
    )

    while True:
        t_sent_str = aud_socket.recv_string()
        data = aud_socket.recv()
        t_recv = time.time()

        # Calculate true audio frame latency by neutralizing the clock drift
        true_latency_ms = (t_recv - float(t_sent_str) + clock_drift) * 1000
        print(f"[Audio Stream] True End-to-End Latency: {true_latency_ms:.1f} ms")

        stream.write(data)


# Run calibration phase prior to launching media loops
calibrate_clock()

threading.Thread(target=receive_audio, daemon=True).start()

vid_socket = context.socket(zmq.SUB)
vid_socket.setsockopt(zmq.RCVHWM, 2)
vid_socket.connect(f"tcp://{SENDER_IP}:5555")
vid_socket.setsockopt(zmq.SUBSCRIBE, b"")

while True:
    t_sent_str = vid_socket.recv_string()
    frame_bytes = vid_socket.recv()
    t_recv = time.time()

    # Calculate true video frame latency by neutralizing the clock drift
    true_latency_ms = (t_recv - float(t_sent_str) + clock_drift) * 1000
    print(f"[Video Stream] True End-to-End Latency: {true_latency_ms:.1f} ms")

    np_arr = np.frombuffer(frame_bytes, np.uint8)
    frame = cv2.imdecode(np_arr, cv2.IMREAD_COLOR)

    if frame is not None:
        cv2.imshow("ZMQ Livestream", frame)

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break

cv2.destroyAllWindows()
sys.exit(0)

```
